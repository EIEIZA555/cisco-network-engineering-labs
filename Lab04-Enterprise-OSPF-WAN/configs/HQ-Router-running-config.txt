! ==========================================================
! HQ-ROUTER RUNNING CONFIGURATION (Lab 04 - OSPF & ROAS)
! ==========================================================
hostname HQ-Router
!
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10
!
ip dhcp pool HR_POOL
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.10.10
!
ip dhcp pool SALES_POOL
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.10.10
!
interface GigabitEthernet0/0/0
 no ip address
 no shutdown
!
interface GigabitEthernet0/0/0.10
 description HR_VLAN_10_GATEWAY
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
!
interface GigabitEthernet0/0/0.20
 description SALES_VLAN_20_GATEWAY
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
!
interface GigabitEthernet0/0/1
 description WAN_LINK_TO_BRANCH
 ip address 10.0.0.1 255.255.255.252
 no shutdown
!
router ospf 1
 router-id 1.1.1.1
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 10.0.0.0 0.0.0.3 area 0
!
end


! ==========================================================
! HQ-SWITCH RUNNING CONFIGURATION (VLANs & 802.1Q Trunk)
! ==========================================================
hostname HQ-Switch
!
vlan 10
 name HR
vlan 20
 name Sales
!
interface GigabitEthernet0/1
 description TRUNK_TO_HQ_ROUTER
 switchport mode trunk
 switchport trunk allowed vlan 10,20
!
interface range FastEthernet0/1 - 10
 description HR_ACCESS_PORTS
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
!
interface range FastEthernet0/11 - 20
 description SALES_ACCESS_PORTS
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
!
end
