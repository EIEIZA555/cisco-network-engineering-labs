! ==========================================================
! CORP-SWITCH RUNNING CONFIGURATION (Lab 05 - Inside LAN)
! ==========================================================
hostname Corp-Switch
!
vlan 10
 name HR_Inside
vlan 20
 name Sales_Inside
!
! --- Dedicated Uplink Ports to ASA Physical Interfaces ---
interface GigabitEthernet0/1
 description UPLINK_TO_ASA_GIG1/7_HR
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
!
interface GigabitEthernet0/2
 description UPLINK_TO_ASA_GIG1/8_SALES
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
!
! --- Workstation Access Ports ---
interface range FastEthernet0/1 - 12
 description HR_WORKSTATIONS
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
!
interface range FastEthernet0/13 - 24
 description SALES_WORKSTATIONS
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
!
end
